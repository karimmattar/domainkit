"""``on_note_read``: the reaction to ``NoteRead``."""

import logging
from uuid import UUID

import pytest
from gen.notes.application import handlers
from gen.notes.domain.events import NoteRead


@pytest.mark.anyio
async def test_on_note_read(caplog: pytest.LogCaptureFixture) -> None:
    """The handler reacts to the event."""
    event = NoteRead(note_id=UUID(int=1))

    with caplog.at_level(logging.INFO, logger=handlers.__name__):
        await handlers.on_note_read(event)

    assert "NoteRead 00000000-0000-0000-0000-000000000001" in caplog.text
