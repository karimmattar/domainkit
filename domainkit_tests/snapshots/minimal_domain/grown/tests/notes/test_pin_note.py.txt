"""``PinNote``: its use case, and its route."""

from datetime import UTC, datetime

import pytest
from gen.notes.application import services
from gen.notes.application.commands import PinNote
from gen.notes.domain.events import NotePinned

from domainkit.ddd import DomainEvent, EventBus
from domainkit.testing import TestClient

from .fakes import FakeNotesUnitOfWork, new_note, new_note_request

URL = "/api/v1/notes/notes"


@pytest.mark.anyio
async def test_pin_note_publishes_note_pinned() -> None:
    """The use case changes the note and commits; NotePinned follows."""
    published: list[DomainEvent] = []
    bus = EventBus()

    async def collect(event: NotePinned) -> None:
        published.append(event)

    bus.subscribe(NotePinned, collect)
    uow = FakeNotesUnitOfWork(bus)
    note = new_note()
    async with uow:
        await uow.notes.add(note)
        await uow.commit()

    await services.pin_note(
        PinNote(note_id=note.id, at=datetime(2026, 1, 2, tzinfo=UTC)), uow
    )

    assert uow.commits == 2
    assert [type(each) for each in published] == [NotePinned]


def test_pin_note_route(admin_client: TestClient) -> None:
    """``POST .../pin`` runs the command on the note the path names."""
    created = admin_client.post(URL, json=new_note_request())
    assert created.status_code == 201, created.json()
    item = created.headers["location"]

    answer = admin_client.post(
        f"{item}/pin",
        json={"at": "2026-01-02T03:04:05Z"},
        headers={"If-Match": created.headers["etag"]},
    )

    assert answer.status_code == 200, answer.json()
    assert answer.json()["id"] == created.json()["id"]
