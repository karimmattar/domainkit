"""HTTP resources of the notes domain (docs/resources.md).

Reads are generic; each change is a command, one use case. Every route
needs its model permission: ``notes.view_note`` to read, and
``add_`` or ``change_`` to change; superusers have them all.
"""

from fastapi import APIRouter

from domainkit.resources import ModelActions, ResourceRouter, schema_for
from gen.notes.application import services
from gen.notes.application.commands import CreateNote, RenameNote
from gen.notes.domain.model import Note
from gen.notes.infrastructure.unit_of_work import SqlAlchemyNotesUnitOfWork

router = APIRouter(tags=["notes"])


async def get_unit_of_work() -> SqlAlchemyNotesUnitOfWork:
    """A unit of work on the database; tests may override this dependency."""
    return SqlAlchemyNotesUnitOfWork()


NoteIn = schema_for(Note, write=True, include=["name"])

notes = ResourceRouter(
    "/notes",
    model=Note,
    unit_of_work=get_unit_of_work,
    repository=lambda uow: uow.notes,
    rule=ModelActions(Note),
)
notes.command_route(CreateNote, service=services.create_note, body=NoteIn)
notes.command_route(
    RenameNote, service=services.rename_note, body=NoteIn, path="/rename"
)
router.include_router(notes)
