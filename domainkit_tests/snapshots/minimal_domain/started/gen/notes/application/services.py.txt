"""Use cases: open a unit of work, change one aggregate, commit."""

from gen.notes.application.commands import CreateNote, RenameNote
from gen.notes.application.unit_of_work import NotesUnitOfWork
from gen.notes.domain.model import Note, NoteId


async def create_note(command: CreateNote, uow: NotesUnitOfWork) -> Note:
    """Create and store a note."""
    note = Note.create(command.name)
    async with uow:
        await uow.notes.add(note)
        await uow.commit()
    return note


async def rename_note(command: RenameNote, uow: NotesUnitOfWork) -> Note:
    """Rename a note."""
    async with uow:
        note = await uow.notes.get(NoteId(command.note_id))
        note.rename(command.name)
        await uow.commit()
    return note
