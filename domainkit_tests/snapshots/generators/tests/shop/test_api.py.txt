"""HTTP tests through the whole stack, with the migrated test database.

They need the domain installed (INSTALLED_DOMAINS) and mounted (config/urls.py).
``admin_client`` calls as a superuser; ``client_for`` as a user with chosen
permissions; ``client`` anonymously. Each test starts from the database as
``migrate`` left it, so it sees only the products it creates.
"""

import io
import logging
from typing import Any
from uuid import uuid4

import pytest

from domainkit.management import call_command
from domainkit.testing import ClientFor, TestClient

URL = "/api/v1/shop/products"
VIEW = "shop.view_product"


def create(client: TestClient, name: str) -> dict[str, Any]:
    """Create a product; return its JSON."""
    created = client.post(URL, json={"name": name})
    assert created.status_code == 201, created.json()
    body: dict[str, Any] = created.json()
    return body


def test_callers_need_the_permissions(
    client: TestClient, client_for: ClientFor
) -> None:
    """Anonymous callers get 401; users without the permission get 403."""
    assert client.get(URL).status_code == 401
    viewer = client_for(VIEW)
    assert viewer.get(URL).status_code == 200
    refused = viewer.post(URL, json={"name": "Not mine"})
    assert (refused.status_code, refused.json()["code"]) == (403, "permission_denied")
    assert client_for().get(URL).status_code == 403


def test_lifecycle(admin_client: TestClient) -> None:
    """Create, read, change on the version read, archive, and delete."""
    created = admin_client.post(URL, json={"name": "  First  "})
    body = created.json()
    item = created.headers["location"]
    assert created.status_code == 201
    assert (body["name"], body["archived"], body["version"], body["notes"]) == (
        "First",
        False,
        1,
        [],
    )
    assert item == f"{URL}/{body['id']}"

    read = admin_client.get(item)
    assert read.json() == body
    assert (
        admin_client.get(
            item, headers={"If-None-Match": read.headers["etag"]}
        ).status_code
        == 304
    )
    renamed = admin_client.patch(
        item, json={"name": "Second"}, headers={"If-Match": read.headers["etag"]}
    )
    assert (renamed.json()["name"], renamed.json()["version"]) == ("Second", 2)
    stale = admin_client.patch(
        item, json={"name": "Third"}, headers={"If-Match": read.headers["etag"]}
    )
    assert (stale.status_code, stale.json()["code"]) == (412, "precondition_failed")

    noted = admin_client.post(f"{item}/notes", json={"text": "Hello"})
    assert [note["text"] for note in noted.json()["notes"]] == ["Hello"]
    archived = admin_client.post(f"{item}/archive")
    assert archived.json()["archived"] is True
    assert admin_client.delete(item).status_code == 204
    assert admin_client.get(item).status_code == 404


def test_business_rules_map_to_http_errors(admin_client: TestClient) -> None:
    """Domain errors become 404/422 responses with a machine-readable code."""
    missing = admin_client.get(f"{URL}/{uuid4()}")
    assert (missing.status_code, missing.json()["code"]) == (404, "product_not_found")

    blank = admin_client.post(URL, json={"name": "   "})
    assert (blank.status_code, blank.json()["code"]) == (422, "invalid_product_name")

    item = f"{URL}/{create(admin_client, 'Frozen')['id']}"
    admin_client.post(f"{item}/archive")
    refused = admin_client.patch(item, json={"name": "Thawed"})
    assert (refused.status_code, refused.json()["code"]) == (422, "product_is_archived")


def test_listings_filter_order_and_page(admin_client: TestClient) -> None:
    """``next_cursor`` walks every page; filters and search narrow them."""
    for name in ("Cherry", "Apple", "Banana"):
        create(admin_client, name)
    admin_client.post(f"{URL}/{create(admin_client, 'Date')['id']}/archive")
    seen: list[str] = []
    params: dict[str, Any] = {"limit": 2, "archived": False, "ordering": "name"}
    while True:
        page = admin_client.get(URL, params=params).json()
        seen += [item["name"] for item in page["items"]]
        if page["next_cursor"] is None:
            break
        params["cursor"] = page["next_cursor"]

    assert seen == ["Apple", "Banana", "Cherry"]
    found = admin_client.get(URL, params={"search": "an", "ordering": "-name"}).json()
    assert [item["name"] for item in found["items"]] == ["Banana"]
    for params in ({"limit": 101}, {"cursor": "bm90LWpzb24"}, {"archived": "maybe"}):
        assert admin_client.get(URL, params=params).status_code == 422


def test_durable_handler_runs_in_the_worker(
    admin_client: TestClient, caplog: pytest.LogCaptureFixture
) -> None:
    """Creating a product queues the durable handler; the worker runs it."""
    created = create(admin_client, "Queued")
    caplog.set_level(logging.INFO, logger="gen.shop.application.handlers")

    call_command("runworker", "--once", stdout=io.StringIO())

    assert f"Product {created['id']} created" in caplog.text
