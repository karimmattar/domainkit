"""Use cases: open a unit of work, change one aggregate, commit.

Committing publishes the events the aggregate recorded. A request's
``If-Match`` is checked as ``get`` loads the aggregate (docs/resources.md).
"""

from domainkit.ddd import ConcurrentUpdate, Retry, Unset
from gen.shop.application.commands import (
    AddProductNote,
    ArchiveProduct,
    CreateCategory,
    CreateProduct,
    CreateWallet,
    DeleteCategory,
    DeleteProduct,
    DeleteWallet,
    LockProduct,
    PublishProduct,
    RetitleCategory,
    TransferWalletBalance,
    UpdateProduct,
)
from gen.shop.application.unit_of_work import ShopUnitOfWork
from gen.shop.domain.model import (
    Category,
    CategoryId,
    OwnerId,
    Product,
    ProductId,
    ProductName,
    Wallet,
    WalletId,
)

RACES = Retry(attempts=5, retry_on=(ConcurrentUpdate,))
"""Up to five runs of a transfer, each after a short random wait."""


async def create_product(command: CreateProduct, uow: ShopUnitOfWork) -> Product:
    """Create and store a product."""
    product = Product.create(ProductName(command.name), OwnerId(command.owner_id))
    async with uow:
        await uow.products.add(product)
        await uow.commit()
    return product


async def update_product(command: UpdateProduct, uow: ShopUnitOfWork) -> Product:
    """Change the fields the command sets."""
    async with uow:
        product = await uow.products.get(ProductId(command.product_id))
        if not isinstance(command.name, Unset):
            product.rename(ProductName(command.name))
        await uow.commit()
    return product


async def archive_product(command: ArchiveProduct, uow: ShopUnitOfWork) -> Product:
    """Archive a product."""
    async with uow:
        product = await uow.products.get(ProductId(command.product_id))
        product.archive()
        await uow.commit()
    return product


async def add_product_note(command: AddProductNote, uow: ShopUnitOfWork) -> Product:
    """Add a note to a product."""
    async with uow:
        product = await uow.products.get(ProductId(command.product_id))
        product.add_note(command.text)
        await uow.commit()
    return product


async def delete_product(command: DeleteProduct, uow: ShopUnitOfWork) -> None:
    """Delete a product; its notes go with it."""
    async with uow:
        product = await uow.products.get(ProductId(command.product_id))
        product.delete()
        await uow.products.remove(product)
        await uow.commit()


async def create_category(command: CreateCategory, uow: ShopUnitOfWork) -> Category:
    """Create and store a category."""
    category = Category.create(
        title=command.title,
        parent=None if command.parent is None else CategoryId(command.parent),
        price=command.price,
    )
    async with uow:
        await uow.categories.add(category)
        await uow.commit()
    return category


async def delete_category(command: DeleteCategory, uow: ShopUnitOfWork) -> None:
    """Delete a category."""
    async with uow:
        category = await uow.categories.get(CategoryId(command.category_id))
        category.delete()
        await uow.categories.remove(category)
        await uow.commit()


async def publish_product(command: PublishProduct, uow: ShopUnitOfWork) -> Product:
    """Publish a product."""
    async with uow:
        product = await uow.products.get(ProductId(command.product_id))
        product.publish(at=command.at, note=command.note)
        await uow.commit()
    return product


async def retitle_category(command: RetitleCategory, uow: ShopUnitOfWork) -> Category:
    """Retitle a category."""
    async with uow:
        category = await uow.categories.get(CategoryId(command.category_id))
        category.retitle(
            title=command.title,
            parent=None if command.parent is None else CategoryId(command.parent),
            rank=command.rank,
        )
        await uow.commit()
    return category


async def lock_product(command: LockProduct, uow: ShopUnitOfWork) -> Product:
    """Lock a product."""
    async with uow:
        product = await uow.products.get(ProductId(command.product_id))
        product.lock()
        await uow.commit()
    return product


async def create_wallet(command: CreateWallet, uow: ShopUnitOfWork) -> Wallet:
    """Create and store a wallet."""
    wallet = Wallet.create(balance=command.balance, owed=command.owed)
    async with uow:
        await uow.wallets.add(wallet)
        await uow.commit()
    return wallet


async def delete_wallet(command: DeleteWallet, uow: ShopUnitOfWork) -> None:
    """Delete a wallet."""
    async with uow:
        wallet = await uow.wallets.get(WalletId(command.wallet_id))
        wallet.delete()
        await uow.wallets.remove(wallet)
        await uow.commit()


async def transfer_wallet_balance(
    command: TransferWalletBalance, uow: ShopUnitOfWork
) -> Wallet:
    """Move the amount from one wallet to another, whole or not at all.

    Both rows are locked in one statement, by ascending id, so two transfers
    between the same wallets take the locks in the same order and cannot
    deadlock; the whole work runs again when its transaction loses a race.

    Add what a project decides: the idempotency middleware, so a request
    retried with its ``Idempotency-Key`` moves the amount once, and a rule
    that the caller may move money out of this wallet
    (``has_object_permission``).
    """
    source, target = WalletId(command.wallet_id), WalletId(command.to)

    async def work() -> Wallet:
        async with uow:
            wallets = await uow.wallets.get_many_for_update([source, target])
            wallets[source].transfer_balance(to=target, amount=command.amount)
            wallets[target].receive_balance(amount=command.amount)
            await uow.commit()
        return wallets[source]

    return await uow.run(work, retry=RACES)
