"""Events."""

from datetime import datetime
from decimal import Decimal
from uuid import UUID

from domainkit.ddd import DomainEvent


class WidgetCreated(DomainEvent):
    """A widget was created."""

    widget_id: UUID


class WidgetDeleted(DomainEvent):
    """A widget was deleted."""

    widget_id: UUID


class TagCreated(DomainEvent):
    """A tag was created."""

    tag_id: UUID


class TagDeleted(DomainEvent):
    """A tag was deleted."""

    tag_id: UUID


class ThingSent(DomainEvent):
    """A thing was sent."""

    thing_id: UUID
    at: datetime
    cost: Decimal
    to: UUID
    code: UUID


class ThingPacked(DomainEvent):
    """A thing was packed."""

    thing_id: UUID


class ThingShipped(DomainEvent):
    """Thing shipped."""

    thing_id: UUID
    at: datetime
    code: UUID
    cost: Decimal
