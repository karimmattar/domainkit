"""``ShipThing``: its use case, and its route."""

from datetime import UTC, datetime
from decimal import Decimal
from uuid import UUID

import pytest
from minimal.things.application import services
from minimal.things.application.commands import ShipThing
from minimal.things.domain.events import ThingSent

from domainkit.ddd import DomainEvent, EventBus
from domainkit.testing import TestClient

from .fakes import FakeThingsUnitOfWork, new_thing, new_thing_request

URL = "/api/v1/things/things"


@pytest.mark.anyio
async def test_ship_thing_publishes_thing_sent() -> None:
    """The use case changes the thing and commits; ThingSent follows."""
    published: list[DomainEvent] = []
    bus = EventBus()

    async def collect(event: ThingSent) -> None:
        published.append(event)

    bus.subscribe(ThingSent, collect)
    uow = FakeThingsUnitOfWork(bus)
    thing = new_thing()
    async with uow:
        await uow.things.add(thing)
        await uow.commit()

    await services.ship_thing(
        ShipThing(
            thing_id=thing.id,
            at=datetime(2026, 1, 2, tzinfo=UTC),
            cost=Decimal("1.50"),
            to=UUID(int=1),
            code=UUID(int=1),
        ),
        uow,
    )

    assert uow.commits == 2
    assert [type(each) for each in published] == [ThingSent]


def test_ship_thing_route(admin_client: TestClient) -> None:
    """``POST .../ship`` runs the command on the thing the path names."""
    created = admin_client.post(URL, json=new_thing_request())
    assert created.status_code == 201, created.json()
    item = created.headers["location"]

    answer = admin_client.post(
        f"{item}/ship",
        json={
            "at": "2026-01-02T03:04:05Z",
            "cost": "1.50",
            "to": "00000000-0000-0000-0000-000000000001",
            "code": "00000000-0000-0000-0000-000000000001",
        },
        headers={"If-Match": created.headers["etag"]},
    )

    assert answer.status_code == 200, answer.json()
    assert answer.json()["id"] == created.json()["id"]
