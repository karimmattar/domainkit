"""The Widget aggregate: its model, its use cases, and its resource."""

from uuid import UUID

import pytest
from minimal.things.application import services
from minimal.things.application.commands import CreateWidget, DeleteWidget
from minimal.things.domain.events import WidgetCreated, WidgetDeleted
from minimal.things.domain.model import ThingId, Widget, WidgetNotFound

from domainkit.testing import TestClient

from .fakes import FakeThingsUnitOfWork, new_widget_request

URL = "/api/v1/things/widgets"


def test_create_and_delete_record_events() -> None:
    """A new widget records WidgetCreated; deleting it, WidgetDeleted."""
    widget = Widget.create(title="text", owner=ThingId(UUID(int=1)), body="text")
    widget.delete()

    created, deleted = widget.pull_events()
    assert isinstance(created, WidgetCreated)
    assert isinstance(deleted, WidgetDeleted)
    assert created.widget_id == deleted.widget_id == widget.id


@pytest.mark.anyio
async def test_use_cases_store_and_remove() -> None:
    """Creating stores and commits; deleting removes; a missing one is not found."""
    uow = FakeThingsUnitOfWork()

    widget = await services.create_widget(
        CreateWidget(title="text", owner=UUID(int=1), body="text"), uow
    )
    assert uow.widgets.items == {widget.id: widget}
    await services.delete_widget(DeleteWidget(widget_id=widget.id), uow)

    assert uow.widgets.items == {}
    with pytest.raises(WidgetNotFound):
        await services.delete_widget(DeleteWidget(widget_id=widget.id), uow)


def test_resource(admin_client: TestClient) -> None:
    """Create, read, list, and delete over HTTP."""
    created = admin_client.post(URL, json=new_widget_request())
    assert created.status_code == 201, created.json()
    item = created.headers["location"]

    assert admin_client.get(item).json() == created.json()
    assert [each["id"] for each in admin_client.get(URL).json()["items"]] == [
        created.json()["id"]
    ]
    assert admin_client.delete(item).status_code == 204
    assert admin_client.get(item).status_code == 404
