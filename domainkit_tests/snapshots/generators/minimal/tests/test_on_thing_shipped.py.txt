"""``on_thing_shipped``: the reaction to ``ThingShipped``."""

import logging
from datetime import UTC, datetime
from decimal import Decimal
from uuid import UUID

import pytest
from minimal.things.application import handlers
from minimal.things.domain.events import ThingShipped


@pytest.mark.anyio
async def test_on_thing_shipped(caplog: pytest.LogCaptureFixture) -> None:
    """The handler reacts to the event."""
    event = ThingShipped(
        thing_id=UUID(int=1),
        at=datetime(2026, 1, 2, tzinfo=UTC),
        code=UUID(int=1),
        cost=Decimal("1.50"),
    )

    with caplog.at_level(logging.INFO, logger=handlers.__name__):
        await handlers.on_thing_shipped(event)

    assert "ThingShipped 00000000-0000-0000-0000-000000000001" in caplog.text
